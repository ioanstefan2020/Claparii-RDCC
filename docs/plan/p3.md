### 3. Contracts & Interfaces

#### 3.1 Browser Storage Schema (`wp.library.v1` in `localStorage`)
```typescript
interface LibraryItem {
  id: string;              // Unique ID, e.g. "lib_1729618200000_a8f2c"
  mode: "simplify" | "plan" | "meeting";
  title: string;           // Derived document title (max ~80 chars)
  createdAt: string;       // ISO 8601 string, e.g. "2026-10-22T18:30:00Z"
  sourceText: string;      // Original input text
  resultData: {            // Direct API response data payload
    summary?: string;
    steps?: Array<{ n: number; text: string; source_quote?: string }>;
    key_terms?: Array<{ term: string; explanation: string }>;
    questions_to_ask?: string[];
    tasks?: Array<{
      id: string;
      title: string;
      due_iso: string | null;
      due_text: string;
      needs_confirmation: boolean;
      priority: string;
      steps: string[];
      source_quote?: string;
    }>;
    decisions?: Array<{ text: string; source_quote?: string }>;
    action_items?: Array<{
      who: string | null;
      task: string;
      due_iso: string | null;
      due_text: string;
      needs_confirmation: boolean;
      source_quote?: string;
    }>;
    unclear_points?: string[];
  };
  tags: string[];          // e.g. ["clarificare", "onboarding"]
}
```

#### 3.2 Storage Module API (`web/js/library-storage.js`)
```javascript
export const LIBRARY_STORAGE_KEY = "wp.library.v1";

/**
 * Returns all saved library items, sorted descending by createdAt.
 * Resilient against missing or malformed storage.
 * @returns {Array<LibraryItem>}
 */
export function getLibraryItems();

/**
 * Saves a new item or updates an existing item.
 * Generates id and createdAt if not provided.
 * @param {Partial<LibraryItem>} item
 * @returns {LibraryItem}
 */
export function saveLibraryItem(item);

/**
 * Deletes an item by id.
 * @param {string} id
 * @returns {LibraryItem | null} The deleted item, or null if not found.
 */
export function deleteLibraryItem(id);

/**
 * Restores a deleted item back into storage (used by Undo).
 * @param {LibraryItem} item
 * @returns {void}
 */
export function restoreLibraryItem(item);

/**
 * Clears all library items.
 * @returns {void}
 */
export function clearLibrary();

/**
 * Returns true if storage is operating on in-memory fallback.
 * @returns {boolean}
 */
export function isLibraryMemoryFallback();
```
